//===== rAthena Script =======================================
//= Ragnadot client test NPC
//===== Description: =========================================
//= A fixture for the Godot client (../ragnarok), not content.
//=
//= Several NPC script primitives cannot be provoked from a GM
//= command — they only happen inside a running script — and the
//= ones that BLOCK are exactly the ones a client must get right:
//= `progressbar` stops the script and sets WIP_DISABLE_ALL, so a
//= client that ignores ZC_PROGRESS soft-locks the player. Every
//= quest script that uses one is behind prerequisites, so this
//= exists to trigger them on demand in prontera.
//=
//= Reload after editing with @reloadscript (no restart needed).
//===== Menu: ================================================
//= Progress bar  -> ZC_PROGRESS, then CZ_PROGRESS after N sec
//=                  (answering EARLY is a cancel, so this also
//=                   tells you which happened)
//= Map marks     -> ZC_COMPASS (viewpoint): a sticky pin, a
//=                  15-second one, a pin MOVED by reusing its
//=                  id, and removal by id
//============================================================

prontera,158,193,4	script	Ragnadot Test	4_M_01,{
	mes "[Ragnadot Test]";
	mes "Client test fixture.";
	next;
	switch (select("Progress bar (3s):Progress bar (8s):Map marks:Remove marks:Clear + align:Cancel")) {
	case 1:
		callsub S_Progress, 3;
		break;
	case 2:
		callsub S_Progress, 8;
		break;
	case 3:
		// Three pins at once, exercising every branch the client must tell apart:
		// a sticky one, a 15-second one, and a THIRD id set twice so the second
		// viewpoint must MOVE it rather than leave two pins behind.
		viewpoint 1, 156, 191, 1, 0xFF0000;   // sticky, red
		viewpoint 0, 150, 180, 2, 0x00FF00;   // 15 seconds, green
		viewpoint 1, 120, 120, 3, 0x0000FF;   // ...first here
		viewpoint 1, 170, 200, 3, 0x0000FF;   // ...then moved, same id
		mes "[Ragnadot Test]";
		mes "Set 3 marks (id 1 sticky, id 2 for 15s, id 3 moved).";
		mes "Expect THREE pins, not four.";
		close;
	case 4:
		viewpoint 2, 0, 0, 1, 0;
		viewpoint 2, 0, 0, 2, 0;
		viewpoint 2, 0, 0, 3, 0;
		mes "[Ragnadot Test]";
		mes "Removed marks 1-3.";
		close;
	case 5:
		// Pile up text, then `clear` it: a client that ignores ZC_CLEAR_DIALOG
		// keeps all six lines and shows the OLDEST, because the box is fixed size.
		mes "LINE ONE - should be gone";
		mes "LINE TWO - should be gone";
		mes "LINE THREE - should be gone";
		next;
		clear;
		setdialogalign 2;   // DIALOG_ALIGN_CENTER
		mes "ONLY THIS LINE should remain, centred.";
		close;
	case 6:
		close;
	}
	end;

// The line after `progressbar` runs ONLY if the client answered at or after the
// server's timeout. An early answer is read as a cancel and ends the script, so
// the two outcomes are distinguishable from the client side.
S_Progress:
	.@secs = getarg(0);
	mes "[Ragnadot Test]";
	mes "Waiting " + .@secs + " seconds...";
	progressbar "0xffff00", .@secs;
	mes "[Ragnadot Test]";
	mes "Wait COMPLETED (not cancelled).";
	close;

// ZC_QUEST_NOTIFY_EFFECT -- the "!"/"?" a player navigates by. `questinfo` is
// re-evaluated on map entry and again on @setquest / @erasequest, so BOTH
// directions can be provoked from a GM command even though the packet itself
// cannot be sent by one:
//
//   @erasequest 1000  -> yellow "!" goes up over this NPC
//   @setquest 1000    -> it comes down, and a purple mark replaces it
//
// Two colours on purpose: the client maps the server's colour enum onto its own
// palette, and one colour would not prove the mapping.
OnInit:
	questinfo( QTYPE_QUEST, QMARK_YELLOW, "isbegin_quest(1000) == 0" );
	questinfo( QTYPE_CLICKME, QMARK_PURPLE, "isbegin_quest(1000) == 1" );
	end;
}
